cmake_minimum_required(VERSION 3.20)
project(san14_enhanced_font LANGUAGES CXX)
set(SAN14_SDK_DIR "" CACHE PATH "SDK checkout containing include/")
if(NOT SAN14_SDK_DIR)
  set(SAN14_SDK_DIR "$ENV{SAN14_SDK_DIR}")
endif()
if(NOT EXISTS "${SAN14_SDK_DIR}/include/san14_mod.h")
  message(FATAL_ERROR "Set SAN14_SDK_DIR to the SDK checkout")
endif()
add_library(enhanced_font SHARED src/enhanced_font.cpp)
target_include_directories(enhanced_font PRIVATE "${SAN14_SDK_DIR}/include")
target_compile_features(enhanced_font PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(enhanced_font PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(enhanced_font_tests tests/enhanced_font_tests.cpp)
  target_include_directories(enhanced_font_tests PRIVATE "${SAN14_SDK_DIR}/include")
  target_compile_features(enhanced_font_tests PRIVATE cxx_std_17)
  add_test(NAME enhanced_font_plugin COMMAND enhanced_font_tests)
endif()
